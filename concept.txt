import torch
import torch.nn as nn
import torch.optim as optim
from torch_geometric.nn import GCNConv
from torch_geometric.data import Data
import pandas as pd
import numpy as np

# Example dataset (replace with your actual data)
# Suppose each row in the dataset corresponds to an edge in the graph, with a label indicating attack type or normal operation
data = pd.read_csv('cyberattack_data.csv')

# Extracting features and labels (Replace these columns with actual ones from your dataset)
X = data[['feature1', 'feature2', 'feature3']].values  # Example feature columns
y = data['attack_type'].values  # Example target column for attack type (0: no attack, 1: attack)

# Convert data to PyTorch tensors
X = torch.tensor(X, dtype=torch.float)
y = torch.tensor(y, dtype=torch.long)

# Create a graph structure (adjacency matrix)
edge_index = torch.tensor([[0, 1, 2], [1, 2, 0]], dtype=torch.long)  # Example graph edges

# Create PyG Data object
data = Data(x=X, edge_index=edge_index, y=y)

# Define the Graph Neural Network model (GCN)
class GNN(nn.Module):
    def __init__(self, in_channels, out_channels):
        super(GNN, self).__init__()
        self.conv1 = GCNConv(in_channels, 16)
        self.conv2 = GCNConv(16, out_channels)
        self.fc = nn.Linear(out_channels, 2)  # 2 classes: attack or no attack

    def forward(self, data):
        x, edge_index = data.x, data.edge_index
        x = self.conv1(x, edge_index)
        x = torch.relu(x)
        x = self.conv2(x, edge_index)
        x = torch.relu(x)
        x = torch.mean(x, dim=0)  # Pooling
        x = self.fc(x)
        return x

# Initialize the model, optimizer, and loss function
model = GNN(in_channels=X.shape[1], out_channels=16)
optimizer = optim.Adam(model.parameters(), lr=0.01)
criterion = nn.CrossEntropyLoss()

# Training loop
for epoch in range(100):
    model.train()
    optimizer.zero_grad()
    out = model(data)  # Forward pass
    loss = criterion(out, data.y)  # Compute the loss
    loss.backward()  # Backpropagation
    optimizer.step()  # Update the weights
    print(f"Epoch {epoch+1}, Loss: {loss.item()}")

# Evaluation (example)
model.eval()
out = model(data)
_, predicted = torch.max(out, 1)
accuracy = (predicted == data.y).sum().item() / len(data.y)
print(f'Accuracy: {accuracy:.4f}')

    "Logistic Regression": LogisticRegression(random_state=42)
}

# Train and evaluate each model
for model_name, model in models.items():
    print(f"Training {model_name}...")
    
    # Fit the model
    model.fit(X_train_scaled, y_train)
    
    # Predict on the test set
    y_pred = model.predict(X_test_scaled)
    
    # Evaluate the model
    print(f"Evaluation for {model_name}:")
    print("Accuracy:", accuracy_score(y_test, y_pred))
    print(classification_report(y_test, y_pred))
